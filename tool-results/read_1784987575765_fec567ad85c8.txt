   535→                  ? "bg-[#1E3A8A] text-white border-[#1E3A8A] dark:bg-[#3B82F6]/15 dark:text-[#3B82F6] dark:border-[#3B82F6]/30"
   536→                  : "bg-card text-muted-foreground border-border hover:bg-muted hover:text-foreground"
   537→              }`}

   541→          ))}
   542→        </div>
   543→      </FilterGroup>
   544→
   545→      {/* Category */}
   546→      <FilterGroup title="Category">

   565→                  ? "bg-[#1E3A8A]/10 text-[#1E3A8A] dark:bg-[#3B82F6]/15 dark:text-[#3B82F6]"

   567→              }`}
   568→            >
   569→              <span className="truncate">{cat}</span>
   570→            </button>
   571→          ))}
   572→        </div>
   573→      </FilterGroup>
   574→
   575→      {/* Difficulty */}
   576→      <FilterGroup title="Difficulty">

   578→          <FilterRadio
   579→            active={difficulty === "ALL"}

   594→
   595→      {/* Scope */}

   598→          <FilterRadio
   599→            active={scope === "ALL"}
   600→            onClick={() => setScope("ALL")}
   601→            label="All Scopes"
   602→          />
   603→          {(["GLOBAL", "REGIONAL", "NATIONAL", "LOCAL"] as ProblemScope[]).map((s) => (
   604→            <FilterRadio
   605→              key={s}
   606→              active={scope === s}

   609→            />
   610→          ))}
   611→        </div>
   612→      </FilterGroup>
   613→
   614→      {/* Project type */}
   615→      <FilterGroup title="Project Type">

   617→          {projectTypeOptions.map((t) => (

   619→              key={t}
   620→              onClick={() => toggleProjectType(t)}
   621→              className={`w-full text-left text-sm px-3 py-2 rounded-lg transition-colors flex items-center gap-2 ${

   623→                  ? "bg-[#1E3A8A]/10 text-[#1E3A8A] dark:bg-[#3B82F6]/15 dark:text-[#3B82F6]"

   631→                    : "border-border"
   632→                }`}
   633→              >
   634→                {projectTypes.includes(t) ? "✓" : ""}
   635→              </span>

   637→            </button>
   638→          ))}
   639→        </div>
   640→      </FilterGroup>
   641→
   642→      {/* Tags */}
   643→      {allTags.length > 0 && (
   644→        <FilterGroup title="Tags">
   645→          <div className="flex flex-wrap gap-1.5 max-h-40 overflow-y-auto sidebar-scroll">
   646→            {allTags.slice(0, 30).map((t) => (
   647→              <button
   648→                key={t}
   649→                onClick={() => toggleTag(t)}
   650→                className={`text-xs px-2 py-1 rounded-md border transition-all ${
   651→                  activeTags.includes(t)
   652→                    ? "bg-[#1E3A8A]/10 text-[#1E3A8A] border-[#1E3A8A]/30 dark:bg-[#3B82F6]/15 dark:text-[#3B82F6] dark:border-[#3B82F6]/30"

   654→                }`}
   655→              >
   656→                {t}
   657→              </button>
   658→            ))}
   659→          </div>
   660→        </FilterGroup>
   661→      )}
   662→
   663→      {/* Footer count */}
   664→      {!isMobile && (
   665→        <div className="pt-2 border-t text-xs text-muted-foreground">
   666→          <span className="text-foreground font-medium">{resultCount}</span> of {totalCount} problems

   669→    </div>
   670→  )
