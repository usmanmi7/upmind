   352→function UserDashboard() {
   353→  const { data: session } = useSession()
   354→  const [data, setData] = React.useState<AnalyticsData | null>(null)
   355→  const [loading, setLoading] = React.useState(true)
   356→
   357→  React.useEffect(() => {
   358→    const fetchData = async () => {
   359→      try {
   360→        const res = await fetch("/api/analytics")
   361→        if (res.ok) {
   362→          const json = await res.json()
   363→          setData(json)
   364→        }
   365→      } catch (error) {
   366→        console.error("Failed to fetch dashboard data:", error)
   367→      } finally {
   368→        setLoading(false)
   369→      }
   370→    }
   371→    fetchData()
   372→  }, [])
   373→
   374→  const firstName = session?.user?.name?.split(" ")[0] || "Founder"
   375→  const overview = data?.overview
   376→  const progress = data?.progress || 0
   377→  const totalTasks = overview?.totalTasks || 0
   378→  const completedTasks = overview?.tasksCompleted || 0
   379→  const taskPercent = totalTasks > 0 ? Math.round((completedTasks / totalTasks) * 100) : 0
   380→
   381→  const formatNextAppointment = () => {
   382→    if (!data?.upcomingAppointments || data.upcomingAppointments.length === 0) return null
   383→    const apt = data.upcomingAppointments[0]
   384→    const date = new Date(apt.date)
   385→    const now = new Date()
   386→    const tomorrow = new Date(now)
   387→    tomorrow.setDate(tomorrow.getDate() + 1)
   388→
   389→    if (date.toDateString() === now.toDateString()) {
   390→      return `Today at ${date.toLocaleTimeString([], { hour: "numeric", minute: "2-digit" })}`
   391→    } else if (date.toDateString() === tomorrow.toDateString()) {
   392→      return `Tomorrow at ${date.toLocaleTimeString([], { hour: "numeric", minute: "2-digit" })}`
   393→    }
   394→    return date.toLocaleDateString([], { month: "short", day: "numeric" }) + ` at ${date.toLocaleTimeString([], { hour: "numeric", minute: "2-digit" })}`
   395→  }
   396→
   397→  if (loading) {
   398→    return (
   399→      <div className="flex items-center justify-center min-h-[400px]">
   400→        <div className="flex flex-col items-center gap-3">
   401→          <Loader2 className="size-8 animate-spin text-[#7CFC00]" />
   402→          <p className="text-sm text-muted-foreground">Loading your dashboard...</p>
   403→        </div>
   404→      </div>
   405→    )
   406→  }
   407→
   408→  return (
   409→    <div className="space-y-6">
   410→      {/* Welcome Section */}
   411→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   412→        <div>
   413→          <h1 className="text-2xl sm:text-3xl font-heading font-bold">
   414→            Welcome back, {firstName} 👋
   415→          </h1>
   416→          <p className="text-muted-foreground mt-1">
   417→            Here&apos;s what&apos;s happening with your startup journey
   418→          </p>
   419→        </div>
   420→        <Button
   421→          asChild
   422→          className="bg-[#7CFC00] hover:bg-[#6BE000] text-[#1A2E1A] shadow-lg shadow-[#7CFC00]/25"
   423→        >
   424→          <Link href="/dashboard/startup">
   425→            <Rocket className="size-4" />
   426→            My Startup

   428→          </Link>
   429→        </Button>
   430→      </div>
   431→
   432→      {/* Stats Cards */}
   433→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   434→        <Card className="border-0 shadow-md shadow-black/5 dark:shadow-black/20">
