    82→/** Generate a short title from the first user message. */
    83→function deriveTitle(text: string): string {
    84→  const clean = text.trim().replace(/\s+/g, " ")
    85→  if (clean.length <= 50) return clean
    86→  const words = clean.split(" ")
    87→  let out = ""
    88→  for (const w of words) {
    89→    if ((out + " " + w).trim().length > 50) break
    90→    out = (out + " " + w).trim()
    91→  }
    92→  return out + "…"
    93→}

   425→                { role: "assistant", content: responseText, structuredJson },
   426→              ],
   427→            }),
   428→          })
   429→          // bump local list
   430→          setChats((prev) =>
   431→            prev.map((c) =>
   432→              c.id === chatId

   434→                    ...c,
   435→                    updatedAt: new Date().toISOString(),
   436→                    messageCount: (c.messageCount ?? 0) + 2,

   450→            prev.map((c) =>
   451→              c.id === chatId

   453→                    ...c,
   454→                    updatedAt: new Date().toISOString(),
   455→                    messageCount: (c.messageCount ?? 0) + 1,

   459→          )

   545→
   546→              {/* Messages (flow with page) */}

   562→                    >
   563→                      <h3 className="text-2xl font-heading font-bold text-foreground">
   564→                        Hi{session?.user?.name ? `, ${session.user.name.split(" ")[0]}` : " there"}!
   565→                      </h3>
   566→                      <p className="text-sm text-muted-foreground mt-2 max-w-md mx-auto leading-relaxed">
