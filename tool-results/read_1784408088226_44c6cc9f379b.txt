    82→    82→/** Generate a short title from the first user message. */
    83→    83→function deriveTitle(text: string): string {
    84→    84→  const clean = text.trim().replace(/\s+/g, " ")
    85→    85→  if (clean.length <= 50) return clean
    86→    86→  const words = clean.split(" ")
    87→    87→  let out = ""
    88→    88→  for (const w of words) {
    89→    89→    if ((out + " " + w).trim().length > 50) break
    90→    90→    out = (out + " " + w).trim()
    91→    91→  }
    92→    92→  return out + "…"
    93→    93→}
