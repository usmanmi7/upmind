   317→                difficulty={difficulty}
   318→                setDifficulty={setDifficulty}
   319→                scope={scope}
   320→                setScope={setScope}
   321→                projectTypes={projectTypes}
   322→                toggleProjectType={toggleProjectType}
   323→                activeTags={activeTags}
   324→                toggleTag={toggleTag}

   351→function FiltersPanel({

   353→  setQuery,
   354→  category,
   355→  setCategory,
   356→  difficulty,
   357→  setDifficulty,
   358→  scope,
   359→  setScope,
   360→  projectTypes,
   361→  toggleProjectType,
   362→  activeTags,
   363→  toggleTag,
   364→  sortBy,
   365→  setSortBy,
   366→  clearAll,
   367→  activeFilterCount,
   368→  categories,
   369→  allTags,
   370→  resultCount,
   371→  totalCount,
   372→  isMobile = false,
   373→}: {
   374→  query: string
   375→  setQuery: (v: string) => void
   376→  category: string
   377→  setCategory: (v: string) => void
   378→  difficulty: DifficultyLevel | "ALL"
   379→  setDifficulty: (v: DifficultyLevel | "ALL") => void
   380→  scope: ProblemScope | "ALL"
   381→  setScope: (v: ProblemScope | "ALL") => void
   382→  projectTypes: string[]
   383→  toggleProjectType: (t: string) => void
   384→  activeTags: string[]
   385→  toggleTag: (t: string) => void
   386→  sortBy: "impactScore" | "severity" | "innovationScore" | "marketNeed" | "futureImportance"
   387→  setSortBy: (
   388→    v: "impactScore" | "severity" | "innovationScore" | "marketNeed" | "futureImportance"
   389→  ) => void
   390→  clearAll: () => void
   391→  activeFilterCount: number
   392→  categories: string[]
   393→  allTags: string[]
   394→  resultCount: number
   395→  totalCount: number
   396→  isMobile?: boolean
   397→}) {

   399→    <div className="space-y-4">

   406→            <span className="text-xs px-2 py-0.5 rounded-full bg-[#7CFC00]/15 text-[#7CFC00] border border-[#7CFC00]/30">

   422→      {/* Search */}

   426→          type="text"
   427→          value={query}

   430→          className="w-full pl-10 pr-3 py-2.5 text-sm bg-white/5 border border-white/10 rounded-lg text-white placeholder:text-white/40 focus:outline-none focus:ring-2 focus:ring-[#7CFC00]/50"

   515→            active={scope === "ALL"}
   516→            onClick={() => setScope("ALL")}
   517→            label="All Scopes"
   518→          />
   519→          {(["GLOBAL", "REGIONAL", "NATIONAL", "LOCAL"] as ProblemScope[]).map((s) => (
   520→            <FilterRadio
   521→              key={s}
   522→              active={scope === s}
   523→              onClick={() => setScope(s)}
